PART 1. 집계 함수 (Aggregate Functions)#

개념: 여러 행(row)을 입력받아 단일 값 하나를 반환하는 함수. GROUP BY, HAVING과 함께 쓰는 것이 핵심 패턴.

COUNT — 행 수 세기#

-- 앨범별 트랙 수 (많은 순)
-- 생각 흐름: "앨범마다 트랙이 몇 개씩?" → AlbumId로 묶고 → TrackId 개수 셈
SELECT AlbumId, COUNT(TrackId) AS track_count
FROM tracks
GROUP BY AlbumId
ORDER BY track_count DESC;

-- DISTINCT 조합: 실제로 구매한 고유 고객 수
-- 생각 흐름: 같은 고객이 여러 번 구매해도 1명으로 세야 함 → DISTINCT 필요
SELECT COUNT(DISTINCT CustomerId) AS 구매고객수
FROM invoices;

SUM — 합계#

-- 연도별 총 매출
-- 생각 흐름: InvoiceDate에서 연도만 뽑아 → 그걸로 묶고 → Total 합산
SELECT strftime('%Y', InvoiceDate) AS 연도,
       ROUND(SUM(Total), 2)        AS 연매출
FROM invoices
GROUP BY 연도
ORDER BY 연도;
-- strftime('%Y', 날짜컬럼) : 날짜에서 연도(4자리)만 추출하는 SQLite 함수

AVG — 평균#

-- 고객 1인당 평균 청구 금액
-- 생각 흐름: invoices 테이블의 Total 전체 평균 → 단순 AVG 한 번으로 끝
SELECT ROUND(AVG(Total), 2) AS 평균청구금액
FROM invoices;

MAX / MIN — 최댓값 / 최솟값#

-- 고객별 최대·최소·평균 청구 금액 한 번에
-- 생각 흐름: 고객마다 여러 청구서가 있음 → CustomerId로 묶고 → 세 값 동시 계산
SELECT CustomerId,
       ROUND(MAX(Total), 2) AS 최대청구,
       ROUND(MIN(Total), 2) AS 최소청구,
       ROUND(AVG(Total), 2) AS 평균청구
FROM invoices
GROUP BY CustomerId
ORDER BY 최대청구 DESC;

GROUP_CONCAT — 문자열 연결#

-- 앨범 1번의 트랙 이름들을 ' / ' 구분자로 한 줄에 연결
-- 생각 흐름: 여러 행의 문자열 값을 하나로 합치고 싶을 때 사용
SELECT GROUP_CONCAT(Name, ' / ') AS 트랙목록
FROM tracks
WHERE AlbumId = 1;
-- 결과 예: "For Those About To Rock / Put The Finger On You / ..."

HAVING — 집계 후 그룹 필터링#

-- WHERE vs HAVING 비교
-- WHERE  : 집계 이전, 개별 행 단위 필터 (집계함수 사용 불가)
-- HAVING : 집계 이후, 그룹 단위 필터  (집계함수 사용 가능)

-- 총 매출이 $40 이상인 고객
-- 생각 흐름: 먼저 고객별로 묶어 SUM → 그 결과에 조건 적용 → HAVING 필요
SELECT CustomerId, ROUND(SUM(Total), 2) AS 총매출
FROM invoices
GROUP BY CustomerId
HAVING 총매출 >= 40
ORDER BY 총매출 DESC;

PART 2. 서브쿼리 심화#

복습: 서브쿼리 = SQL 안에 또 다른 SELECT. 위치에 따라 역할이 달라집니다.

WHERE 절 서브쿼리 — 조건 비교#

-- 평균 청구 금액보다 높은 청구서 목록
-- 생각 흐름:
--   1단계(서브쿼리): 전체 평균 Total 값을 먼저 구한다
--   2단계(메인쿼리): 그 값보다 큰 행만 가져온다
SELECT InvoiceId, CustomerId, ROUND(Total, 2) AS 청구금액
FROM invoices
WHERE Total > (
    SELECT AVG(Total)  -- ← 숫자 1개 반환 → > 연산자 사용 가능
    FROM invoices
)
ORDER BY Total DESC;

-- NOT IN: 한 번도 판매되지 않은 트랙
-- 생각 흐름:
--   1단계(서브쿼리): invoice_items에 등장한 TrackId 목록 수집
--   2단계(메인쿼리): 그 목록에 없는 트랙만 조회
-- ⚠️ 주의: 서브쿼리 결과에 NULL이 있으면 NOT IN은 아무것도 반환 안 할 수 있음
--           → 안전한 방법은 아래 NOT EXISTS 사용
SELECT TrackId, Name
FROM tracks
WHERE TrackId NOT IN (
    SELECT DISTINCT TrackId  -- DISTINCT로 중복 제거
    FROM invoice_items
)
ORDER BY TrackId;

FROM 절 서브쿼리 — 인라인 뷰 (파생 테이블)#

-- 고객별 총 매출 TOP 5
-- 생각 흐름:
--   1단계(서브쿼리): 고객별 총매출을 먼저 계산해서 임시 테이블(sub)을 만듦
--   2단계(메인쿼리): 그 임시 테이블을 customers와 JOIN해서 고객 이름까지 붙임
-- ⚠️ FROM 절 서브쿼리는 반드시 별칭(AS sub) 필수!
SELECT sub.CustomerId,
       c.FirstName || ' ' || c.LastName AS 고객명,
       sub.총매출
FROM (
    SELECT CustomerId, ROUND(SUM(Total), 2) AS 총매출
    FROM invoices
    GROUP BY CustomerId       -- ← 여기서 먼저 집계
) AS sub                      -- ← 이 결과가 임시 테이블이 됨
JOIN customers c ON sub.CustomerId = c.CustomerId
ORDER BY sub.총매출 DESC
LIMIT 5;

연관 서브쿼리 (Correlated Subquery) — 행마다 서브쿼리 재실행#

-- 자신이 속한 장르의 평균 단가보다 비싼 트랙
-- 생각 흐름:
--   외부 쿼리가 트랙 하나(t)를 읽을 때마다
--   → 서브쿼리가 "이 트랙의 장르(t.GenreId)의 평균 단가"를 그때그때 계산
--   → 자기 장르 평균보다 비싼 트랙만 통과
-- ⚠️ 행마다 서브쿼리가 실행되므로 데이터가 많으면 느릴 수 있음
SELECT t.Name, t.UnitPrice, g.Name AS 장르
FROM tracks t
JOIN genres g ON t.GenreId = g.GenreId
WHERE t.UnitPrice > (
    SELECT AVG(t2.UnitPrice)
    FROM tracks t2
    WHERE t2.GenreId = t.GenreId  -- ← 외부 쿼리의 t.GenreId를 참조 (연관!)
)
ORDER BY g.Name, t.UnitPrice DESC;

EXISTS / NOT EXISTS — 존재 여부 확인#

-- 개념: IN은 값 목록 비교 / EXISTS는 "해당 행이 존재하냐?"만 확인
-- EXISTS는 서브쿼리가 행을 하나라도 반환하면 TRUE
-- NOT EXISTS는 NOT IN보다 NULL에 안전

-- 한 번도 판매된 적 없는 트랙 (NOT EXISTS 버전)
-- 생각 흐름:
--   각 트랙(t)마다 → invoice_items에 그 트랙이 있는지 검사
--   → 없으면(NOT EXISTS) 해당 트랙을 결과에 포함
SELECT t.TrackId, t.Name
FROM tracks t
WHERE NOT EXISTS (
    SELECT 1              -- ← 어떤 값을 반환하든 상관없음, 존재 여부만 봄
    FROM invoice_items ii
    WHERE ii.TrackId = t.TrackId  -- ← 연관: 외부 t.TrackId 참조
);

PART 3. 집합 연산자 (Set Operators)#

개념: 두 SELECT 결과를 집합처럼 합치거나 빼는 연산. 공통 규칙:

  • 두 SELECT의 컬럼 수가 같아야 한다
  • 대응 컬럼의 데이터 타입이 호환되어야 한다
  • 최종 컬럼명은 첫 번째 쿼리 기준
  • ORDER BY는 맨 마지막 한 번만
연산자 중복 제거 의미
UNION O 합집합
UNION ALL X (빠름) 합집합 (중복 포함)
INTERSECT O 교집합
EXCEPT O 차집합

UNION — 합집합 (중복 제거)#

-- 직원 이름 + 고객 이름 통합 목록
-- 생각 흐름:
--   두 테이블에서 각각 이름을 뽑아 → 세로로 합치기 (중복 자동 제거)
--   → 어느 쪽인지 구분하기 위해 'Type' 컬럼 추가
SELECT FirstName, LastName, 'Employee' AS Type
FROM employees
UNION
SELECT FirstName, LastName, 'Customer'
FROM customers
ORDER BY FirstName, LastName;
-- ⚠️ ORDER BY는 맨 마지막에 단 한 번만

UNION ALL — 합집합 (중복 포함, 소계 패턴에 유용)#

-- 연도별 매출 + 전체 합계 리포트
-- 생각 흐름:
--   위쪽 쿼리: 연도별 소계를 구함
--   아래쪽 쿼리: 전체 합계를 한 행으로 추가
--   → UNION ALL로 합쳐서 마치 "소계 + 합계" 리포트처럼 만듦
SELECT strftime('%Y', InvoiceDate) AS 연도,
       ROUND(SUM(Total), 2)        AS 매출합계
FROM invoices
GROUP BY 연도

UNION ALL  -- 중복 제거 없이 그냥 붙임 (UNION보다 빠름)

SELECT '── 전체합계',
       ROUND(SUM(Total), 2)
FROM invoices

ORDER BY 연도;

INTERSECT — 교집합#

-- 2009년과 2010년 모두 구매한 고객 ID
-- 생각 흐름:
--   위쪽: 2009년에 구매한 고객 집합
--   아래쪽: 2010년에 구매한 고객 집합
--   → 교집합(INTERSECT) = 두 해 모두 구매한 고객만 남음
SELECT CustomerId
FROM invoices
WHERE strftime('%Y', InvoiceDate) = '2009'

INTERSECT

SELECT CustomerId
FROM invoices
WHERE strftime('%Y', InvoiceDate) = '2010'

ORDER BY CustomerId;

EXCEPT — 차집합#

-- 2009년 구매 고객 중 2010년에는 구매하지 않은 고객
-- 생각 흐름:
--   위쪽(A): 2009년 구매 고객 집합
--   아래쪽(B): 2010년 구매 고객 집합
--   → A에서 B를 뺌(EXCEPT) = 2009년엔 샀지만 2010년엔 안 산 고객
SELECT CustomerId
FROM invoices
WHERE strftime('%Y', InvoiceDate) = '2009'

EXCEPT

SELECT CustomerId
FROM invoices
WHERE strftime('%Y', InvoiceDate) = '2010'

ORDER BY CustomerId;

PART 4. 계층형 질의 (Hierarchical Query)#

개념: 데이터 자체가 부모-자식 구조(트리)일 때 사용. SQLite에서는 WITH RECURSIVE CTE(공통 테이블 표현식)로 구현합니다.

대상: employees 테이블

  • EmployeeId: 직원 PK
  • ReportsTo: 상위 관리자 FK (NULL = 최상위 사장님)

WITH RECURSIVE — 기본 계층 조회#

-- 직원 전체 보고 체계 조회
-- 생각 흐름:
--   1단계(Anchor): 출발점 설정 → ReportsTo가 NULL인 최상위 직원 1명
--   2단계(Recursive): "이 직원의 부하 → 그 부하의 부하 → ..."를 반복
--   3단계(출력): lvl(계층 깊이)을 이용해 들여쓰기로 시각화

WITH RECURSIVE emp_hierarchy(
    EmployeeId, FirstName, LastName, Title, ReportsTo, lvl
) AS (
    -- [Anchor] 출발점: 최상위 직원 (사장)
    SELECT EmployeeId, FirstName, LastName, Title, ReportsTo,
           0 AS lvl          -- 최상위 = 레벨 0
    FROM employees
    WHERE ReportsTo IS NULL  -- ← 여기서 시작

    UNION ALL                -- 재귀는 반드시 UNION ALL

    -- [Recursive] 반복: 위에서 찾은 직원의 부하 직원을 계속 탐색
    SELECT e.EmployeeId, e.FirstName, e.LastName,
           e.Title, e.ReportsTo,
           h.lvl + 1         -- ← 레벨을 1씩 증가
    FROM employees e
    JOIN emp_hierarchy h
      ON e.ReportsTo = h.EmployeeId  -- ← "내 상사 = 직전 단계에서 찾은 직원"
)
-- [출력] 계층 레벨만큼 공백으로 들여쓰기
SELECT lvl AS 계층레벨,
       SUBSTR('                ', 1, lvl * 4)
           || FirstName || ' ' || LastName AS 직원명,
       Title AS 직책
FROM emp_hierarchy
ORDER BY lvl, EmployeeId;
-- 결과 예:
-- 0 | Andrew Adams         | General Manager
-- 1 |     Nancy Edwards    | Sales Manager
-- 2 |         Jane Peacock | Sales Support Agent

WITH RECURSIVE + PATH — 조직 경로 추적#

-- 각 직원의 "사장 → 부장 → 직원" 경로 문자열 생성
-- 생각 흐름:
--   Anchor에서 경로 시작값(이름) 설정
--   → Recursive 단계마다 " > 이름"을 문자열 뒤에 계속 붙임
--   → 마지막에 노드 유형(Leaf=부하없음, Branch=관리자)도 판별

WITH RECURSIVE emp_path(..., path) AS (
    -- Anchor: 경로의 시작 = 최상위 직원 이름
    SELECT ..., FirstName || ' ' || LastName AS path
    FROM employees
    WHERE ReportsTo IS NULL

    UNION ALL

    -- Recursive: 경로에 " > 부하이름" 계속 추가
    SELECT ..., p.path || ' > ' || e.FirstName || ' ' || e.LastName
    FROM employees e
    JOIN emp_path p ON e.ReportsTo = p.EmployeeId
)
SELECT path AS 조직경로,
       -- Leaf 판별: 내 EmployeeId가 누군가의 ReportsTo에 없으면 = 부하가 없음
       CASE
           WHEN EmployeeId NOT IN (
               SELECT ReportsTo FROM employees WHERE ReportsTo IS NOT NULL
           ) THEN 'Leaf'    -- 말단 직원
           ELSE 'Branch'    -- 중간 관리자
       END AS 노드유형
FROM emp_path
ORDER BY path;
-- 결과 예:
-- Andrew Adams > Nancy Edwards > Jane Peacock | Leaf
-- Andrew Adams > Nancy Edwards > Steve Johnson | Leaf

🗂️ 오늘 배운 것 한눈에 보기#

구분 키워드 핵심
집계 함수 COUNT / SUM / AVG / MAX / MIN 여러 행 → 단일 값
GROUP_CONCAT 여러 문자열 → 한 줄 연결
HAVING 집계 후 그룹 필터 (WHERE와 다름)
서브쿼리 WHERE 절 조건 값을 동적으로 계산
FROM 절 (인라인 뷰) 집계 결과를 임시 테이블로 활용
연관 서브쿼리 외부 쿼리 값을 참조, 행마다 재실행
EXISTS / NOT EXISTS NULL 안전한 존재 여부 확인
집합 연산자 UNION 합집합 (중복 제거)
UNION ALL 합집합 (중복 포함, 소계 패턴)
INTERSECT 교집합 (두 집합에 모두 있는 것)
EXCEPT 차집합 (A에서 B를 뺀 것)
계층형 WITH RECURSIVE 트리 구조 데이터 탐색
