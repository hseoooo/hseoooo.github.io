Python 클래스(Class) 개념 요약#

IT 회사 팀원/부서 예시로 익히는 클래스의 모든 개념

1. 클래스(Class)란?#

클래스는 객체를 만들기 위한 설계도(청사진) 다.

클래스(Class)    = 설계도       → "팀원"이라는 개념 자체
인스턴스(Instance) = 실제 물건  → 실제로 존재하는 "Alice", "Bob"

클래스 안에는 두 가지가 들어간다.

  • 속성(Attribute): 객체가 가지는 데이터 (이름, 직책, 연봉 등)
  • 메서드(Method): 객체가 할 수 있는 행동 (자기소개, 프로젝트 합류 등)

가장 단순한 클래스#

# class 키워드 + 클래스 이름 (첫 글자는 대문자가 관례)
class TeamMember:
    pass  # 아직 내용이 없는 빈 클래스

# 클래스로부터 인스턴스(객체) 생성
member1 = TeamMember()  # TeamMember 설계도로 member1이라는 실제 객체를 만든다
member2 = TeamMember()  # 또 다른 독립적인 객체

print(member1 is member2)  # False → 같은 설계도로 만들었지만 서로 다른 객체

2. __init__ — 생성자 (인스턴스 초기화)#

__init__은 인스턴스가 생성되는 순간 자동으로 딱 한 번 실행되는 특별한 메서드다. 주로 인스턴스의 초기 속성(데이터)을 설정하는 데 사용한다.

self란?#

self는 “지금 이 메서드를 호출하고 있는 인스턴스 자신” 을 가리킨다. alice.introduce() 를 호출하면, self는 alice가 된다.

class TeamMember:

    def __init__(self, name, role, salary):
        # self.속성명 = 값  →  이 인스턴스만의 데이터를 저장
        # 왼쪽 self.name : 이 인스턴스에 name이라는 속성을 만든다
        # 오른쪽 name    : __init__이 호출될 때 전달받은 인자값
        self.name = name
        self.role = role
        self.salary = salary
        self.projects = []  # 속성의 초깃값을 직접 지정할 수도 있다 (빈 리스트)

# 인스턴스 생성: 괄호 안에 __init__의 self를 제외한 인자를 전달
alice = TeamMember(name="Alice", role="시니어 개발자", salary=6000)
#                  ↑ self는 자동으로 alice가 됨, 직접 전달하지 않는다

# 속성 접근: 인스턴스.속성명
print(alice.name)    # Alice
print(alice.salary)  # 6000

3. 인스턴스 메서드#

클래스 내부에 정의된 함수. 첫 번째 매개변수는 반드시 self 다. self를 통해 인스턴스의 속성을 읽거나 수정할 수 있다.

class TeamMember:

    def __init__(self, name, role, salary):
        self.name = name
        self.role = role
        self.salary = salary
        self.projects = []

    def introduce(self):
        # self.role, self.name → 이 인스턴스의 속성에 접근
        print(f"안녕하세요! 저는 {self.role} {self.name}입니다.")

    def join_project(self, project_name):
        # self.projects → 이 인스턴스의 리스트 속성에 항목 추가
        self.projects.append(project_name)
        print(f"{self.name}이(가) '{project_name}' 프로젝트에 합류했습니다.")

    def show_projects(self):
        if self.projects:  # 리스트가 비어있지 않으면
            print(f"{self.name}의 현재 프로젝트: {', '.join(self.projects)}")
        else:
            print(f"{self.name}은(는) 현재 참여 중인 프로젝트가 없습니다.")

    def get_raise(self, amount):
        # self.salary += amount → 이 인스턴스의 salary 속성값을 변경
        self.salary += amount
        print(f"{self.name}의 연봉이 {amount}만원 인상되어 {self.salary}만원이 되었습니다.")


alice = TeamMember("Alice", "시니어 개발자", 6000)
alice.introduce()
alice.join_project("AI 챗봇 개발")   # alice.projects = ["AI 챗봇 개발"]
alice.show_projects()
alice.get_raise(500)                  # alice.salary = 6500

4. 클래스 변수 vs 인스턴스 변수#

구분 설명 선언 위치 접근 방법
클래스 변수 모든 인스턴스가 공유하는 변수 클래스 내부, __init__ 바깥 클래스명.변수명 또는 self.변수명
인스턴스 변수 각 인스턴스마다 독립적인 변수 __init__ 내부, self.변수명 인스턴스.변수명 
class TeamMember:
    # ── 클래스 변수: 클래스 바로 아래에 선언, 모든 인스턴스가 공유 ──
    company_name = "TechCorp"  # 회사명은 모두 동일
    total_members = 0          # 전체 팀원 수 카운터

    def __init__(self, name, role, salary):
        # ── 인스턴스 변수: self.변수명으로 선언, 각 인스턴스마다 다른 값 ──
        self.name = name
        self.role = role
        self.salary = salary

        # 새 팀원이 생성될 때마다 클래스 변수를 업데이트
        # self.total_members += 1 이 아니라 클래스명.변수명으로 접근해야 함
        TeamMember.total_members += 1

    def introduce(self):
        # 클래스 변수는 self.company_name으로도 접근 가능
        print(f"[{self.company_name}] {self.role} {self.name}")


alice = TeamMember("Alice", "시니어 개발자", 6000)
bob   = TeamMember("Bob",   "기획자", 3800)

print(TeamMember.total_members)  # 2 → 클래스 변수는 클래스명으로 접근

# 클래스 변수를 바꾸면 → 모든 인스턴스에 즉시 반영
TeamMember.company_name = "TechCorp Korea"
alice.introduce()  # [TechCorp Korea] 시니어 개발자 Alice
bob.introduce()    # [TechCorp Korea] 기획자 Bob

5. __str__과 __repr__ — 객체를 문자열로 표현#

정의하지 않으면 print(alice) 시 <__main__.TeamMember object at 0x...> 처럼 알아보기 힘든 형태로 출력된다.

메서드 호출 시점 용도
__str__ print(객체), str(객체) 사람이 읽기 좋은 형태
__repr__ repr(객체), 디버깅 콘솔 개발자용, 객체를 재현할 수 있는 정보 
class TeamMember:
    def __init__(self, name, role, salary):
        self.name = name
        self.role = role
        self.salary = salary

    def __str__(self):
        # print(alice) 또는 str(alice) 호출 시 이 값이 출력됨
        return f"{self.role} {self.name}"

    def __repr__(self):
        # repr(alice) 호출 시, 또는 리스트 안에 넣고 출력할 때 사용됨
        return f"TeamMember(name='{self.name}', role='{self.role}', salary={self.salary})"


alice = TeamMember("Alice", "시니어 개발자", 6000)

print(alice)        # 시니어 개발자 Alice        ← __str__ 호출
print(repr(alice))  # TeamMember(name='Alice', ...)  ← __repr__ 호출

6. 클래스 메서드 & 정적 메서드#

종류 데코레이터 첫 번째 매개변수 주요 용도
인스턴스 메서드 없음 self (인스턴스) 인스턴스 데이터 읽기/수정
클래스 메서드 @classmethod cls (클래스 자체) 클래스 변수 접근, 대체 생성자
정적 메서드 @staticmethod 없음 클래스와 관련은 있지만 self/cls 불필요한 유틸리티 
class TeamMember:
    company_name = "TechCorp"
    total_members = 0

    def __init__(self, name, role, salary):
        self.name = name
        self.role = role
        self.salary = salary
        TeamMember.total_members += 1

    # ── 클래스 메서드: cls = 클래스 자체 ──
    @classmethod
    def get_total_members(cls):
        # cls.total_members = TeamMember.total_members 와 동일
        return cls.total_members

    @classmethod
    def from_string(cls, member_string):
        """
        대체 생성자(Alternative Constructor)
        "Alice-시니어 개발자-6000" 형식의 문자열로 인스턴스를 생성
        """
        name, role, salary = member_string.split("-")
        # cls(...)는 TeamMember(...)와 동일 → 새 인스턴스를 반환
        return cls(name, role, int(salary))

    # ── 정적 메서드: self도 cls도 받지 않음 ──
    @staticmethod
    def is_valid_salary(salary):
        """연봉 유효성 검사 (1000 ~ 200000만원)"""
        # self나 cls가 필요 없는 순수한 기능 — 그냥 함수처럼 동작
        return 1000 <= salary <= 200000

    def __str__(self):
        return f"{self.role} {self.name} (연봉: {self.salary}만원)"


alice = TeamMember("Alice", "시니어 개발자", 6000)
bob   = TeamMember("Bob",   "기획자", 3800)

# 클래스 메서드 호출: 인스턴스 없이 클래스명으로 직접 호출
print(TeamMember.get_total_members())  # 2

# 대체 생성자로 인스턴스 생성
carol = TeamMember.from_string("Carol-데이터 분석가-4500")
print(carol)  # 데이터 분석가 Carol (연봉: 4500만원)

# 정적 메서드 호출
print(TeamMember.is_valid_salary(6000))  # True
print(TeamMember.is_valid_salary(500))   # False

7. 상속(Inheritance)#

기존 클래스(부모)의 속성과 메서드를 물려받아 새로운 클래스(자식)를 만드는 것. 공통 기능은 부모에 두고, 팀마다 다른 기능은 자식에 추가한다.

Department (부모 클래스 — 모든 팀의 공통 기능)
    ├── DevTeam       (개발팀)
    ├── PlanningTeam  (기획팀)
    ├── SalesTeam     (영업팀)
    └── AnalyticsTeam (분석팀)

부모 클래스#

class Department:
    """모든 팀이 공통으로 가지는 속성과 행동"""

    def __init__(self, team_name, budget):
        self.team_name = team_name
        self.budget = budget
        self.members = []  # TeamMember 인스턴스들이 담길 리스트

    def add_member(self, member):
        # member는 TeamMember 인스턴스 — 인스턴스를 리스트에 저장할 수 있다
        self.members.append(member)
        print(f"[{self.team_name}] {member.name}님이 합류했습니다.")

    def show_members(self):
        print(f"\n=== {self.team_name} 구성원 ({len(self.members)}명) ===")
        for i, member in enumerate(self.members, 1):
            print(f"  {i}. {member.role}: {member.name} (연봉 {member.salary}만원)")

    def get_total_salary(self):
        return sum(m.salary for m in self.members)

    def team_meeting(self):
        # 자식 클래스에서 이 메서드를 재정의(override)할 수 있음
        print(f"[{self.team_name}] 정기 팀 회의를 시작합니다.")

자식 클래스 — super()와 오버라이딩#

class DevTeam(Department):  # 괄호 안에 부모 클래스 이름을 적어 상속
    """개발팀 — Department를 상속받아 개발팀 전용 기능을 추가"""

    def __init__(self, budget, tech_stack):
        # super().__init__() → 부모 클래스의 __init__을 호출
        # 부모가 처리하는 team_name, budget, members 초기화를 그대로 실행
        super().__init__("개발팀", budget)

        # 부모에는 없는 개발팀만의 추가 속성
        self.tech_stack = tech_stack

    def show_tech_stack(self):
        """개발팀 전용 메서드 — 부모에는 없는 기능"""
        print(f"[{self.team_name}] 기술 스택: {', '.join(self.tech_stack)}")

    def code_review(self):
        print(f"[{self.team_name}] 코드 리뷰 시작. 참여자: {', '.join(m.name for m in self.members)}")

    # 오버라이딩(Overriding): 부모의 team_meeting()을 개발팀 방식으로 재정의
    def team_meeting(self):
        # 같은 이름의 메서드지만, 개발팀만의 내용으로 완전히 새로 작성
        print(f"[{self.team_name}] 스프린트 계획 회의 (기술 스택: {', '.join(self.tech_stack)})")

class SalesTeam(Department):

    def __init__(self, budget, monthly_target):
        super().__init__("영업팀", budget)
        self.monthly_target = monthly_target
        self.monthly_achievement = 0  # 달성액은 0에서 시작

    def close_deal(self, amount):
        self.monthly_achievement += amount
        rate = (self.monthly_achievement / self.monthly_target) * 100
        print(f"[{self.team_name}] 계약 성사! +{amount}만원 (달성률: {rate:.1f}%)")

    def team_meeting(self):
        rate = (self.monthly_achievement / self.monthly_target) * 100
        print(f"[{self.team_name}] 영업 현황 회의 (목표: {self.monthly_target}, 달성: {self.monthly_achievement}, {rate:.1f}%)")

8. 다형성(Polymorphism)#

같은 이름의 메서드를 호출해도 어떤 클래스의 인스턴스냐에 따라 다르게 동작하는 것. 오버라이딩을 통해 자연스럽게 구현된다.

# 서로 다른 클래스의 인스턴스들을 하나의 리스트에 담아 같은 메서드를 호출
all_teams = [dev_team, planning_team, sales_team, analytics_team]

for team in all_teams:
    # team이 어떤 클래스인지에 따라 각자 다른 team_meeting()이 실행됨
    # → DevTeam이면 스프린트 회의, SalesTeam이면 영업 현황 회의 등
    team.team_meeting()

9. isinstance()와 issubclass() — 타입 확인#

# isinstance(객체, 클래스): 객체가 해당 클래스(또는 그 자식 클래스)의 인스턴스인지 확인
print(isinstance(dev_team, DevTeam))     # True  → dev_team은 DevTeam 인스턴스
print(isinstance(dev_team, Department))  # True  → DevTeam은 Department를 상속하므로 True
print(isinstance(alice, Department))     # False → alice는 TeamMember 인스턴스

# issubclass(자식, 부모): 상속 관계인지 확인 (인스턴스가 아닌 클래스 자체를 전달)
print(issubclass(DevTeam, Department))   # True
print(issubclass(SalesTeam, DevTeam))    # False → 형제 관계, 부모-자식 아님

실습 연습 코드 — 도서관 시스템#

위에서 배운 모든 개념(클래스, 인스턴스, 상속, 오버라이딩, 다형성, 클래스 변수, 클래스 메서드, 정적 메서드, __str__)을 한 번에 연습하는 코드다.

# ══════════════════════════════════════════════════════════
#  도서관 시스템 — 클래스 종합 실습
#
#  구조:
#    Book (책 — 기본 클래스)
#      ├── EBook    (전자책)
#      └── AudioBook (오디오북)
#
#    Library (도서관)
# ══════════════════════════════════════════════════════════


# ────────────────────────────────────────────────────────
# 1. Book 클래스 — 모든 책의 공통 설계도
# ────────────────────────────────────────────────────────
class Book:
    # ── 클래스 변수: 모든 Book 인스턴스가 공유 ──
    total_books = 0       # 지금까지 생성된 책의 총 수
    library_name = "중앙 도서관"

    def __init__(self, title, author, year):
        # ── 인스턴스 변수: 각 책마다 다른 데이터 ──
        self.title = title
        self.author = author
        self.year = year
        self.is_borrowed = False  # 초기 대출 상태는 False (대출 가능)

        # 책이 생성될 때마다 클래스 변수 카운터 증가
        Book.total_books += 1

    # ── 인스턴스 메서드: self를 통해 이 책의 데이터를 읽거나 수정 ──
    def borrow(self):
        """책을 대출하는 메서드"""
        if not self.is_borrowed:
            # self.is_borrowed → 이 책의 대출 상태를 True로 변경
            self.is_borrowed = True
            print(f"'{self.title}' 대출 완료!")
        else:
            print(f"'{self.title}'은(는) 이미 대출 중입니다.")

    def return_book(self):
        """책을 반납하는 메서드"""
        if self.is_borrowed:
            self.is_borrowed = False
            print(f"'{self.title}' 반납 완료!")
        else:
            print(f"'{self.title}'은(는) 대출 중이 아닙니다.")

    def get_info(self):
        """책 정보를 반환하는 메서드 — 자식 클래스에서 오버라이딩 가능"""
        status = "대출 중" if self.is_borrowed else "대출 가능"
        return f"[{status}] 《{self.title}》 — {self.author} ({self.year})"

    # ── 클래스 메서드: cls를 통해 클래스 변수에 접근 ──
    @classmethod
    def get_total_books(cls):
        """지금까지 생성된 전체 책 수 반환"""
        return cls.total_books

    @classmethod
    def from_string(cls, book_string):
        """
        대체 생성자: "제목/저자/출판연도" 형식 문자열로 책 생성
        예) "파이썬 완전정복/홍길동/2024"
        """
        title, author, year = book_string.split("/")
        # cls(...)는 Book(...) 또는 자식 클래스(...)와 동일
        return cls(title, author, int(year))

    # ── 정적 메서드: self도 cls도 필요 없는 독립적인 유틸리티 ──
    @staticmethod
    def is_valid_year(year):
        """출판연도가 유효한 범위(1900~2030)인지 확인"""
        return 1900 <= year <= 2030

    # ── 매직 메서드: print(book) 시 출력 형태 지정 ──
    def __str__(self):
        return f"《{self.title}》 ({self.author})"

    def __repr__(self):
        return f"Book(title='{self.title}', author='{self.author}', year={self.year})"

# 2. EBook 클래스 — Book을 상속
# ────────────────────────────────────────────────────────
class EBook(Book):  # 괄호 안에 부모 클래스를 적어 상속

    def __init__(self, title, author, year, file_size_mb):
        # super().__init__() → 부모(Book)의 __init__ 실행
        # title, author, year, is_borrowed, total_books 증가 등을 부모에게 위임
        super().__init__(title, author, year)

        # 전자책만의 추가 속성
        self.file_size_mb = file_size_mb

    # ── 오버라이딩: 부모의 get_info()를 전자책 방식으로 재정의 ──
    def get_info(self):
        status = "대출 중" if self.is_borrowed else "대출 가능"
        # 부모와 같은 이름이지만, 전자책 정보(용량)를 추가로 출력
        return f"[{status}][E-Book] 《{self.title}》 — {self.author} ({self.year}) | {self.file_size_mb}MB"

    def download(self):
        """전자책 전용 메서드"""
        print(f"《{self.title}》 다운로드 중... ({self.file_size_mb}MB)")


# ────────────────────────────────────────────────────────
# 3. AudioBook 클래스 — Book을 상속
# ────────────────────────────────────────────────────────
class AudioBook(Book):

    def __init__(self, title, author, year, duration_min):
        super().__init__(title, author, year)

        # 오디오북만의 추가 속성
        self.duration_min = duration_min

    # ── 오버라이딩: 오디오북 방식으로 재정의 ──
    def get_info(self):
        status = "대출 중" if self.is_borrowed else "대출 가능"
        hours, minutes = divmod(self.duration_min, 60)
        return f"[{status}][Audio] 《{self.title}》 — {self.author} ({self.year}) | {hours}시간 {minutes}분"

    def play(self):
        """오디오북 전용 메서드"""
        print(f"《{self.title}》 재생 중... (총 {self.duration_min}분)")

# 4. Library 클래스 — Book 인스턴스들을 관리
# ────────────────────────────────────────────────────────
class Library:

    def __init__(self, name):
        self.name = name
        self.books = []  # Book 인스턴스들이 담길 리스트

    def add_book(self, book):
        """책을 도서관에 추가 — book은 Book(또는 자식) 인스턴스"""
        self.books.append(book)
        print(f"[{self.name}] {book} 추가 완료")

    def show_all(self):
        """보유한 모든 책 목록 출력"""
        print(f"\n{'='*50}")
        print(f"  {self.name} 보유 도서 목록 (총 {len(self.books)}권)")
        print(f"{'='*50}")
        for i, book in enumerate(self.books, 1):
            # book.get_info() → Book이면 Book의 get_info(),
            #                    EBook이면 EBook의 get_info() 실행 (다형성!)
            print(f"  {i}. {book.get_info()}")

    def search(self, keyword):
        """제목 또는 저자명으로 책 검색"""
        results = [b for b in self.books if keyword in b.title or keyword in b.author]
        if results:
            print(f"\n검색 결과 '{keyword}' ({len(results)}건):")
            for book in results:
                print(f"  - {book}")
        else:
            print(f"'{keyword}'에 해당하는 책이 없습니다.")

    def __str__(self):
        return f"{self.name} (보유 {len(self.books)}권)"
